cmake_minimum_required(VERSION 3.18)
project(d4cnn_cpu LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
find_package(pybind11 CONFIG REQUIRED)
find_package(Threads REQUIRED)
set(ANACAL_SOURCE_DIR "" CACHE PATH "AnaCal checkout for native q-stamp support")
add_library(d4cnn_core STATIC native/convolution.cpp native/kernels.cpp native/kernels_simd.cpp)
target_include_directories(d4cnn_core PUBLIC include)
target_compile_options(d4cnn_core PRIVATE -O3 -Wall -Wextra)
target_link_libraries(d4cnn_core PUBLIC Threads::Threads)
include(CTest)
if(BUILD_TESTING)
  add_executable(test_kernels tests/test_kernels.cpp)
  target_link_libraries(test_kernels PRIVATE d4cnn_core)
  target_compile_options(test_kernels PRIVATE -O3)
  add_test(NAME kernels COMMAND test_kernels)
  set_tests_properties(kernels PROPERTIES SKIP_RETURN_CODE 77)
endif()
pybind11_add_module(_native NO_EXTRAS native/bindings.cpp)
target_link_libraries(_native PRIVATE d4cnn_core)
target_compile_options(_native PRIVATE -O3 -Wall -Wextra)
if(ANACAL_SOURCE_DIR)
  if(NOT EXISTS "${ANACAL_SOURCE_DIR}/include/anacal/image.h")
    message(FATAL_ERROR "ANACAL_SOURCE_DIR must contain include/anacal/image.h")
  endif()
  target_sources(_native PRIVATE native/anacal_adapter.cpp "${ANACAL_SOURCE_DIR}/src/image.cpp")
  target_include_directories(_native PRIVATE "${ANACAL_SOURCE_DIR}/include")
  target_compile_definitions(_native PRIVATE D4CNN_WITH_ANACAL=1)
endif()
install(TARGETS _native LIBRARY DESTINATION cpu_src)

# Install Python explicitly: a pre-existing in-tree .so must not overwrite the
# extension just compiled by CMake when building a wheel from a working checkout.
install(FILES __init__.py __main__.py api.py model_io.py export.py reference.py
  make_golden.py validation.py science_validation.py DESTINATION cpu_src)
install(DIRECTORY pipeline benchmarks DESTINATION cpu_src
  FILES_MATCHING PATTERN "*.py" PATTERN "__pycache__" EXCLUDE)
install(DIRECTORY configs DESTINATION cpu_src FILES_MATCHING PATTERN "*.json")
install(FILES README.md DESTINATION cpu_src)
